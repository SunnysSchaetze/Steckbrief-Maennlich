/* =========================================================
   3D-MAUSBEWEGUNG DER KARTE

const card = document.getElementById("card");

card.addEventListener("mousemove", (e) => {

    const rect = card.getBoundingClientRect();

    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;

    const centerX = rect.width / 2;
    const centerY = rect.height / 2;

    const rotateY = ((x - centerX) / centerX) * 8;
    const rotateX = ((centerY - y) / centerY) * 8;

    card.style.transform =
        `rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale(1.025)`;
});

card.addEventListener("mouseleave", () => {

    card.style.transform =
        "rotateX(0deg) rotateY(0deg) scale(1)";

});


/* =========================================================
   CURSOR-GLITZEREFFEKT
   ========================================================= */

let letzteX = 0;
let letzteY = 0;
let letzterSpawn = 0;
let aktiveGlitzer = 0;

const MAX_GLITZER = 180;

document.addEventListener(
    "mousemove",
    function(event) {

        erzeugeBewegungsEffekt(
            event.clientX,
            event.clientY
        );

    }
);

document.addEventListener(
    "touchmove",
    function(event) {

        if (!event.touches || !event.touches.length) {
            return;
        }

        const touch = event.touches[0];

        erzeugeBewegungsEffekt(
            touch.clientX,
            touch.clientY
        );

    },
    {
        passive: true
    }
);

document.addEventListener(
    "click",
    function(event) {

        klickEffekt(
            event.clientX,
            event.clientY
        );

    }
);


function erzeugeBewegungsEffekt(x, y) {

    const dx = x - letzteX;
    const dy = y - letzteY;

    const entfernung = Math.sqrt(
        dx * dx +
        dy * dy
    );

    const jetzt = performance.now();

    if (entfernung < 2) {
        return;
    }

    if (jetzt - letzterSpawn < 12) {

        letzteX = x;
        letzteY = y;

        return;
    }

    if (aktiveGlitzer >= MAX_GLITZER) {

        letzteX = x;
        letzteY = y;

        return;
    }

    const nx = dx / entfernung;
    const ny = dy / entfernung;

    const trailX = x;
    const trailY = y;

    createGlitzer(
        trailX,
        trailY,
        false,
        nx,
        ny
    );

    createStaub(trailX, trailY);
    createStaub(trailX, trailY);

    createStaub(
        trailX - nx * 6,
        trailY - ny * 6
    );

    createStaub(
        trailX - nx * 12,
        trailY - ny * 12
    );

    createStaub(
        trailX - nx * 25,
        trailY - ny * 25
    );

    createStaub(
        trailX - nx * 40,
        trailY - ny * 40
    );

    if (Math.random() > .68) {

        createGlitzer(
            trailX,
            trailY,
            true,
            nx,
            ny
        );

    }

    if (entfernung > 15) {

        createStaub(
            trailX,
            trailY
        );

        createStaub(
            trailX - nx * 15,
            trailY - ny * 15
        );

        createStaub(
            trailX - nx * 30,
            trailY - ny * 30
        );

        if (entfernung > 28) {

            createStaub(
                trailX - nx * 45,
                trailY - ny * 45
            );

            createStaub(
                trailX - nx * 60,
                trailY - ny * 60
            );

        }
    }

    letzterSpawn = jetzt;

    letzteX = x;
    letzteY = y;
}


function createGlitzer(
    x,
    y,
    grosser = false,
    nx = 0,
    ny = 0
) {

    if (aktiveGlitzer >= MAX_GLITZER) {
        return;
    }

    const element =
        document.createElement("div");

    element.className =
        grosser
            ? "glitzer long"
            : "glitzer";

    const color =
        Math.random() > .5
            ? "#ff8500"
            : "#ffe000";

    const size =
        grosser
            ? 10 + Math.random() * 5
            : 4 + Math.random() * 4;

    element.style.setProperty(
        "--color",
        color
    );

    element.style.setProperty(
        "--size",
        size + "px"
    );

    const seitlich =
        (Math.random() - .5) * 5;

    const px = -ny;
    const py = nx;

    const sx = px * seitlich;
    const sy = py * seitlich;

    const mx =
        sx -
        nx * (16 + Math.random() * 16);

    const my =
        sy -
        ny * (16 + Math.random() * 16);

    const ex =
        mx -
        nx * (32 + Math.random() * 35) +
        (Math.random() - .5) * 5;

    const ey =
        my -
        ny * (32 + Math.random() * 35) +
        (Math.random() - .5) * 5;

    element.style.setProperty(
        "--sx",
        sx + "px"
    );

    element.style.setProperty(
        "--sy",
        sy + "px"
    );

    element.style.setProperty(
        "--mx",
        mx + "px"
    );

    element.style.setProperty(
        "--my",
        my + "px"
    );

    element.style.setProperty(
        "--ex",
        ex + "px"
    );

    element.style.setProperty(
        "--ey",
        ey + "px"
    );

    const duration =
        900 + Math.random() * 550;

    element.style.setProperty(
        "--duration",
        duration + "ms"
    );

    element.style.left =
        x + "px";

    element.style.top =
        y + "px";

    aktiveGlitzer++;

    document.body.appendChild(element);

    setTimeout(() => {

        element.remove();

        aktiveGlitzer--;

    }, duration + 60);
}
